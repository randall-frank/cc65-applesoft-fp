171void page1()
172{
173 int err, cmp, sgn;
174 AS_FAC_FP ram_half = {0x80, {0x00, 0x00, 0x00, 0x00}}; /* 0.5 */
175 AS_FAC_FP ram_temp;
176
177 title();
178
179 asfp_mem2fac(&AS_CONST_two_pi);
180 asfp_mem_mul_fac(&AS_CONST_half);
181 printf("Computed PI: %s\n", asfp_fac2str());
182
183 asfp_str2fac("1.234e-5");
184 printf("Parsed from '1.234e-5': %s\n", asfp_fac2str());
185
186 err = asfp_str2fac("Invalid");
187 if (err) printf("'Invalid' correctly returned an error.\n");
188 err = asfp_str2fac("2.0");
189 if (err) printf("Error parsing: '2.0' : %d\n", err);
190
191 asfp_mem2fac(&ram_half);
192 asfp_mem_pow_fac(&AS_CONST_two);
193 printf("Computed pow(2.0, 0.5): %s\n", asfp_fac2str());
194 asfp_mem2fac(&AS_CONST_sqrt_two);
195 printf("Const sqrt(2): %s\n", asfp_fac2str());
196
197 // Comparison
198 asfp_str2fac("0.4");
199 cmp = asfp_fac_cmp_mem(&ram_half);
200 printf("Compare 0.4 (FAC) to 0.5 (mem): %d\n", cmp);
201 asfp_mem2fac(&AS_CONST_half);
202 cmp = asfp_fac_cmp_mem(&ram_half);
203 printf("Compare 0.5 (FAC) to 0.5 (mem): %d\n", cmp);
204
205 // 'sign' functions
206 asfp_str2fac("-1002.4");
207 sgn = asfp_sgn();
208 printf("C sign check: %d (%s)\n", sgn, asfp_fac2str());
209 asfp_sgn_fac();
210 printf("FAC sign check: %s\n", asfp_fac2str());
211
212 // Alternative FAC load routines
213 asfp_char2fac(0xff);
214 printf("Signed byte load: %s\n", asfp_fac2str());
215 asfp_uchar2fac(0xff);
216 printf("Unsigned byte load: %s\n", asfp_fac2str());
217 asfp_int2fac(-30000);
218 printf("Int load: %s\n", asfp_fac2str());
219
220 // Random number (Note: first number in the sequence)
221 asfp_rnd_fac();
222 printf("Random number: %s\n", asfp_fac2str());
223}
